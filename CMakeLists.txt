cmake_minimum_required(VERSION 3.22)
option(SNIFFER_HOST "Build portable core and tests without ESP-IDF" OFF)
if(SNIFFER_HOST)
    project(surveillance_hound VERSION 0.1.0 LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 20)
    enable_testing()
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
else()
    set(IDF_TARGET esp32)
    set(COMPONENTS main)
    include($ENV{IDF_PATH}/tools/cmake/project.cmake)
    project(surveillance_hound VERSION 0.1.0)
    execute_process(COMMAND git -C "$ENV{IDF_PATH}" rev-parse HEAD OUTPUT_VARIABLE SDK_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT SDK_COMMIT STREQUAL "7101770dc6db2667b3c477cc31365dd1acd6db4e")
        message(FATAL_ERROR "ESP-IDF v6.0.2 commit 7101770dc6db2667b3c477cc31365dd1acd6db4e required")
    endif()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_target(validate_data ALL
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/compile_signatures.py --check
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/pack_assets.py --check
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
if(SNIFFER_HOST)
    add_library(sniffer_core components/sniffer_core/parsers.cpp components/sniffer_core/json.cpp components/sniffer_core/utilities.cpp components/sniffer_core/engine.cpp

    target_include_directories(sniffer_core PUBLIC components/sniffer_core/include generated)
    target_link_libraries(sniffer_core PUBLIC OpenSSL::Crypto)
    target_compile_options(sniffer_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
    option(SNIFFER_SANITIZE "Enable address and undefined behavior sanitizers" ON)
    if(SNIFFER_SANITIZE)
        target_compile_options(sniffer_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(sniffer_core PUBLIC -fsanitize=address,undefined)
    endif()
    add_dependencies(sniffer_core validate_data)
    add_library(sniffer_history components/storage/history_export.cpp components/storage/ignore_backup.cpp components/storage/history_recovery.cpp)
    target_include_directories(sniffer_history PUBLIC components/storage/include)
    target_link_libraries(sniffer_history PUBLIC sniffer_core)
    target_compile_options(sniffer_history PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 13 AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 14)
        # GCC 13's analyzer reports leaks at fopen/opendir despite the explicit
        # cleanup and handles retained between recovery steps. Keep diagnostics
        # visible; storage tests verify closure on completion/cancel/I/O failure.
        set_source_files_properties(components/storage/history_export.cpp components/storage/history_recovery.cpp
            PROPERTIES COMPILE_OPTIONS "-Wno-error=analyzer-file-leak;-Wno-error=analyzer-malloc-leak")
    endif()
    add_executable(storage_export_tests tests/host/storage_export_tests.cpp)
    target_link_libraries(storage_export_tests PRIVATE sniffer_history)
    target_link_options(storage_export_tests PRIVATE
        -Wl,--wrap=fopen -Wl,--wrap=fclose -Wl,--wrap=fsync -Wl,--wrap=fflush)
    add_test(NAME storage_export COMMAND storage_export_tests)
    add_executable(history_recovery_tests tests/host/history_recovery_tests.cpp)
    target_link_libraries(history_recovery_tests PRIVATE sniffer_history)
    target_link_options(history_recovery_tests PRIVATE
        -Wl,--wrap=fopen -Wl,--wrap=fclose -Wl,--wrap=fsync -Wl,--wrap=ftruncate)
    add_test(NAME history_recovery COMMAND history_recovery_tests)
    add_executable(ignore_backup_tests tests/host/ignore_backup_tests.cpp)
    target_link_libraries(ignore_backup_tests PRIVATE sniffer_history)
    target_link_options(ignore_backup_tests PRIVATE
        -Wl,--wrap=fopen -Wl,--wrap=fclose -Wl,--wrap=fsync -Wl,--wrap=fflush -Wl,--wrap=fwrite)
    add_test(NAME ignore_backup COMMAND ignore_backup_tests)
    add_executable(storage_state_tests tests/host/storage_state_tests.cpp components/storage/state_store.cpp)
    target_include_directories(storage_state_tests PRIVATE tests/host/nvs_fake components/storage/include components/board_e32r40t/include)
    target_link_libraries(storage_state_tests PRIVATE sniffer_core)
    target_compile_options(storage_state_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME storage_state COMMAND storage_state_tests)
    add_executable(core_tests tests/host/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE sniffer_core)
    add_test(NAME core COMMAND core_tests)
    add_test(NAME tools COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests/host -p "test_*.py")
    set_tests_properties(tools PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
    add_executable(sniffer_demo tools/demo.cpp)
    target_link_libraries(sniffer_demo PRIVATE sniffer_core)
    add_executable(fuzz_smoke tests/fuzz/fuzz_smoke.cpp)
    target_link_libraries(fuzz_smoke PRIVATE sniffer_core)
    add_test(NAME fuzz_smoke COMMAND fuzz_smoke)
    add_executable(fuzz_mutate tests/fuzz/mutate.cpp tests/fuzz/libfuzzer.cpp)
    target_link_libraries(fuzz_mutate PRIVATE sniffer_core)
    add_library(sniffer_ui components/ui/ui.cpp)
    target_include_directories(sniffer_ui PUBLIC components/ui/include)
    target_link_libraries(sniffer_ui PUBLIC sniffer_core)
    target_compile_options(sniffer_ui PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_executable(ui_preview tools/preview.cpp)
    target_link_libraries(ui_preview PRIVATE sniffer_ui)
    add_executable(simulator_worker tools/simulator_worker.cpp tools/simulator.cpp)
    target_link_libraries(simulator_worker PRIVATE sniffer_ui)
    target_compile_options(simulator_worker PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_executable(ui_tests tests/host/ui_tests.cpp)
    target_link_libraries(ui_tests PRIVATE sniffer_ui)
    target_include_directories(ui_tests PRIVATE components/storage/include components/board_e32r40t/include)
    add_test(NAME ui COMMAND ui_tests)
    add_executable(feature_tests tests/host/feature_tests.cpp)
